% Le titre et la référence du livre du chapitre restent accessibles à l'image.
\DeclareDataInheritance{inbook}{misc}{%
    \inherit{title}{usera}
    \inherit{crossref}{userb}
}
\DeclareDataInheritance{book}{misc}{%
    \inherit{title}{booktitle}
}

% Conserver le pilote du style courant pour les autres entrées @misc.
\csletcs{blx@bbx@miscstandard}{blx@bbx@misc}
\DeclareBibliographyDriver{misc}{%
    \iffieldequalstr{entrysubtype}{imageextrait}
        {%
            \iffieldundef{booktitle}
                {\PackageError{biblio}{Titre du livre absent pour \thefield{entrykey}}{Vérifier le crossref du chapitre vers un livre avec un titre.}}
                {}%
            \iffieldundef{userb}
                {\iffieldundef{crossref}
                    {\PackageError{biblio}{Référence du document absente pour \thefield{entrykey}}{Ajouter un crossref vers le document source.}}
                    {}}%
                {}%
            \usebibmacro{bibindex}%
            \usebibmacro{begentry}%
            \printnames{author}%
            \setunit{\addcomma\space}%
            \printfield{title}%
            \setunit{\addcomma\space}%
            \printtext{extrait de}%
            \setunit{\addspace}%
            \usebibmacro{booktitle}%
            \setunit{\addspace}%
            \iffieldundef{userb}
                {\printtext{\cite{\thefield{crossref}}}}
                {\printtext{\cite{\thefield{userb}}}}%
            \iffieldundef{usera}
                {}
                {\setunit{\addcomma\space}%
                 \printfield{usera}}%
            \setunit{\addcomma\space}%
            \printfield{pages}%
            \newunit\newblock
            \usebibmacro{bytranslator+others}%
            \newunit\newblock
            \usebibmacro{addendum+pubstate}%
            \usebibmacro{finentry}%
        }
        {\csuse{blx@bbx@miscstandard}}%
}

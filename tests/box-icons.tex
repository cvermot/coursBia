\newcommand{\checkicon}[3]{%
  \ifluatex
    {\FontAwsome\iffontchar\font#1\relax\else
      \errmessage{Missing Font Awesome 7 glyph #1}%
    \fi}%
  \fi
  \setbox0=\hbox{#2}%
  \setbox2=\hbox{#3}%
  \ifdim\wd0=\wd2\else\errmessage{Icon width changed}\fi
  \ifdim\ht0=\ht2\else\errmessage{Icon height changed}\fi
  \ifdim\dp0=\dp2\else\errmessage{Icon depth changed}\fi
  \box2
}
